-- قلاع النفود v4: friends, chat, presence, friendly battles
alter table public.players add column if not exists code text;
alter table public.players add column if not exists seen_at timestamptz not null default now();
update public.players set code = upper(substr(md5(id::text || 'nf'), 1, 6)) where code is null;
create unique index if not exists players_code_idx on public.players(code);

create table if not exists public.friends(
  a uuid not null references public.players(id) on delete cascade,
  b uuid not null references public.players(id) on delete cascade,
  at timestamptz not null default now(),
  primary key(a, b)
);
alter table public.friends enable row level security;
revoke all on public.friends from anon, authenticated;

create table if not exists public.messages(
  id bigserial primary key,
  room text not null,
  sender uuid references public.players(id) on delete cascade,
  sender_name text not null,
  body text not null,
  at timestamptz not null default now()
);
create index if not exists messages_room_idx on public.messages(room, id desc);
alter table public.messages enable row level security;
revoke all on public.messages from anon, authenticated;

create or replace function public.nf_seen(p_id uuid) returns void language sql security definer set search_path = public as $$
  update players set seen_at = now(), code = coalesce(code, upper(substr(md5(id::text || 'nf'), 1, 6))) where id = p_id and seen_at < now() - interval '20 seconds';
$$;
revoke all on function public.nf_seen(uuid) from public, anon, authenticated;

create or replace function public.nf_me(p_id uuid, p_secret text)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  perform nf_seen(p_id);
  select * into r from players where id = p_id;
  return json_build_object('code', r.code, 'trophies', r.trophies, 'online', (select count(*) from players where seen_at > now() - interval '3 minutes'));
end $$;

create or replace function public.nf_friend_add(p_id uuid, p_secret text, p_code text)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; o players; n int;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  select * into o from players where code = upper(trim(p_code));
  if o.id is null then raise exception 'no such code'; end if;
  if o.id = p_id then raise exception 'self'; end if;
  select count(*) into n from friends where a = p_id; if n >= 50 then raise exception 'too many'; end if;
  insert into friends(a, b) values (p_id, o.id), (o.id, p_id) on conflict do nothing;
  return json_build_object('name', o.name);
end $$;

create or replace function public.nf_friend_del(p_id uuid, p_secret text, p_friend uuid)
returns void language plpgsql security definer set search_path = public, extensions as $$
declare r players;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  delete from friends where (a = p_id and b = p_friend) or (a = p_friend and b = p_id);
end $$;

create or replace function public.nf_friends(p_id uuid, p_secret text)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; L json;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  perform nf_seen(p_id);
  select coalesce(json_agg(json_build_object('id', p.id, 'name', p.name, 'trophies', p.trophies, 'castle', p.castle,
      'online', p.seen_at > now() - interval '3 minutes', 'seen', extract(epoch from now() - p.seen_at)::int,
      'unread', (select count(*) from messages m where m.room = 'dm:' || least(p_id::text, p.id::text) || ':' || greatest(p_id::text, p.id::text) and m.sender = p.id and m.at > now() - interval '2 days'))
      order by p.seen_at desc), '[]'::json) into L
  from friends f join players p on p.id = f.b where f.a = p_id;
  return L;
end $$;

-- rooms: 'g' (everyone) or 'dm' with a friend
create or replace function public.nf_room(p_id uuid, p_room text) returns text language plpgsql stable security definer set search_path = public as $$
declare f uuid;
begin
  if p_room = 'g' then return 'g'; end if;
  if p_room like 'dm:%' then
    f := substr(p_room, 4)::uuid;
    if not exists(select 1 from friends where a = p_id and b = f) then raise exception 'not friends'; end if;
    return 'dm:' || least(p_id::text, f::text) || ':' || greatest(p_id::text, f::text);
  end if;
  raise exception 'bad room';
end $$;
revoke all on function public.nf_room(uuid, text) from public, anon, authenticated;

create or replace function public.nf_chat_send(p_id uuid, p_secret text, p_room text, p_body text)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; rm text; b text; n int;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  rm := nf_room(p_id, p_room);
  b := left(regexp_replace(trim(coalesce(p_body, '')), '\s+', ' ', 'g'), 200);
  if char_length(b) < 1 then raise exception 'empty'; end if;
  select count(*) into n from messages where sender = p_id and at > now() - interval '10 seconds';
  if n >= 4 then raise exception 'slow down'; end if;
  insert into messages(room, sender, sender_name, body) values (rm, p_id, r.name, b);
  delete from messages where room = rm and id < (select coalesce(min(id), 0) from (select id from messages where room = rm order by id desc limit 300) x);
  perform nf_seen(p_id);
  return json_build_object('ok', true);
end $$;

create or replace function public.nf_chat_get(p_id uuid, p_secret text, p_room text, p_after bigint)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; rm text; L json;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  rm := nf_room(p_id, p_room);
  perform nf_seen(p_id);
  select coalesce(json_agg(x order by x.id), '[]'::json) into L from (
    select m.id, m.sender = p_id as me, m.sender as sid, m.sender_name as name, m.body, extract(epoch from m.at)::bigint as t
    from messages m where m.room = rm and m.id > coalesce(p_after, 0) order by m.id desc limit 60) x;
  return json_build_object('items', L, 'online', (select count(*) from players where seen_at > now() - interval '3 minutes'));
end $$;

-- friendly battle against a friend's base: verified like any battle, no trophies change hands
create or replace function public.nf_ticket_friend(p_id uuid, p_secret text, p_friend uuid)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; o players; bt battles; sd int;
begin
  r := nf_auth(p_id, p_secret); if r.id is null then raise exception 'no player'; end if;
  if not exists(select 1 from friends where a = p_id and b = p_friend) then raise exception 'not friends'; end if;
  select * into o from players where id = p_friend;
  if o.base is null then raise exception 'no base'; end if;
  if exists(select 1 from battles where attacker = p_id and created_at > now() - interval '1 second') then raise exception 'slow down'; end if;
  update battles set status = 'abandoned' where attacker = p_id and status in ('open','started') and kind = 'friend';
  sd := 1 + floor(random() * 2000000000)::int;
  insert into battles(attacker, defender, kind, seed, lvl, base, opp_name, opp_trophies)
    values (p_id, o.id, 'friend', sd, greatest(1, least(8, o.castle)), o.base->'b', o.name, o.trophies) returning * into bt;
  return json_build_object('tid', bt.id, 'seed', sd, 'L', bt.lvl, 'kind', 'friend', 'pen', 0,
    'opp', json_build_object('name', o.name, 'trophies', o.trophies, 'castle', o.castle, 'base', o.base->'b'));
end $$;

-- claim/settle/inbox understand friendly battles
create or replace function public.nf_claim(p_tid uuid, p_id uuid, p_secret text)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; b battles;
begin
  r := nf_auth(p_id, p_secret);
  if r.id is null then raise exception 'no player'; end if;
  select * into b from battles where id = p_tid and attacker = p_id for update;
  if b.id is null then raise exception 'no ticket'; end if;
  if b.status not in ('open','started') then raise exception 'ticket used'; end if;
  if b.created_at < now() - interval '40 minutes' then raise exception 'ticket expired'; end if;
  update battles set status = 'verifying' where id = p_tid;
  return json_build_object('tid', b.id, 'seed', b.seed, 'L', b.lvl, 'kind', b.kind, 'opts', coalesce(b.opts, '{}'::jsonb), 'att_castle', r.castle,
    'opp', case when b.kind in ('pvp','friend') then json_build_object('name', b.opp_name, 'trophies', b.opp_trophies, 'castle', b.lvl, 'base', b.base) else null end);
end $$;

create or replace function public.nf_settle(p_tid uuid, p_res jsonb, p_replay jsonb)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare b battles; r players; st int; pc int; tr int; td int := 0; hr int; dy int; lg int; lw int;
begin
  select * into b from battles where id = p_tid for update;
  if b.id is null or b.status <> 'verifying' then raise exception 'bad ticket'; end if;
  st := greatest(0, least(3, coalesce((p_res->>'stars')::int, 0)));
  pc := greatest(0, least(100, coalesce((p_res->>'pct')::int, 0)));
  select * into r from players where id = b.attacker;
  if b.kind = 'friend' then
    insert into attacks_log(defender, attacker, attacker_name, battle, stars, pct, tr_att, tr_def, loot_g, loot_w, replay)
      values (b.defender, b.attacker, r.name, b.id, st, pc, 0, 0, 0, 0, p_replay);
    update battles set status = 'done', result = p_res where id = p_tid;
    return json_build_object('tr', 0, 'trophies', r.trophies, 'tr_def', 0, 'stars', st, 'pct', pc, 'friendly', true);
  end if;
  if st > 0 then tr := st * 8 + (b.seed % 5);
    select coalesce(sum(delta),0) into hr from trophy_events where player_id = b.attacker and at > now() - interval '1 hour' and delta > 0;
    select coalesce(sum(delta),0) into dy from trophy_events where player_id = b.attacker and at > now() - interval '1 day' and delta > 0;
    tr := greatest(0, least(tr, 400 - hr, 2000 - dy));
  else tr := -(6 + (b.seed % 7)); end if;
  insert into trophy_events(player_id, nonce, kind, delta) values (b.attacker, 'b:' || b.id::text, 'verified', tr) on conflict do nothing;
  update players set trophies = greatest(0, trophies + tr), updated_at = now() where id = b.attacker returning * into r;
  if b.kind = 'pvp' and b.defender is not null then
    td := case when st > 0 then -greatest(1, round(abs(tr) * 0.75)::int) else 3 + (b.seed % 4) end;
    lg := least(coalesce((p_res->>'gotG')::int, 0), 200000); lw := least(coalesce((p_res->>'gotW')::int, 0), 200000);
    update players set trophies = greatest(0, trophies + td),
      shield_until = case when st >= 3 then now() + interval '45 minutes' when st > 0 then now() + interval '20 minutes' else shield_until end
      where id = b.defender;
    insert into attacks_log(defender, attacker, attacker_name, battle, stars, pct, tr_att, tr_def, loot_g, loot_w, replay)
      values (b.defender, b.attacker, r.name, b.id, st, pc, tr, td, lg, lw, p_replay);
    delete from attacks_log where defender = b.defender and at < now() - interval '7 days';
  end if;
  update battles set status = 'done', result = p_res where id = p_tid;
  delete from battles where attacker = b.attacker and created_at < now() - interval '2 days';
  return json_build_object('tr', tr, 'trophies', r.trophies, 'tr_def', td, 'stars', st, 'pct', pc);
end $$;

create or replace function public.nf_inbox(p_id uuid, p_secret text)
returns json language plpgsql security definer set search_path = public, extensions as $$
declare r players; items json;
begin
  r := nf_auth(p_id, p_secret);
  if r.id is null then raise exception 'no player'; end if;
  perform nf_seen(p_id);
  select coalesce(json_agg(x order by x.at desc), '[]'::json) into items from (
    select a.id, a.attacker_name, a.stars, a.pct, a.tr_att, a.tr_def, a.loot_g, a.loot_w, a.replay, a.at, b.kind = 'friend' as friendly,
      (select castle from players where id = a.attacker) as att_castle,
      json_build_object('tid', b.id, 'seed', b.seed, 'L', b.lvl, 'kind', b.kind, 'opts', '{}'::jsonb,
        'opp', json_build_object('name', b.opp_name, 'trophies', b.opp_trophies, 'castle', b.lvl, 'base', b.base)) as t
    from attacks_log a join battles b on b.id = a.battle
    where a.defender = p_id and not a.seen order by a.at desc limit 10) x;
  update attacks_log set seen = true where defender = p_id and not seen;
  return json_build_object('items', items, 'trophies', r.trophies);
end $$;

-- new players get a friend code
create or replace function public.nf_code_trg() returns trigger language plpgsql as $$
begin new.code := coalesce(new.code, upper(substr(md5(new.id::text || 'nf'), 1, 6))); return new; end $$;
drop trigger if exists players_code on public.players;
create trigger players_code before insert on public.players for each row execute function public.nf_code_trg();
